# Windows NT 4.0 Workstation (the SP1 CD) onto an empty NT4.vhd, then
# Service Pack 6 from its CD, up to the desktop with SP6. For
# session_86box.py on TestVM as configured: vpc2007 (AMI BIOS 8), Pentium II
# 333, 128 MB, S3 Vision964 (Stealth 64 Video), the CD before the hard disk
# in its boot order. Before each run, a fresh disk in the VM directory:
# rm NT4.vhd && truncate -s 4G NT4.vhd
#
# Setup's choices: Custom, every component and sub-component, name and
# computer name TestVM, no Administrator password, no Emergency Repair Disk,
# not on a network, time zone (GMT) without daylight saving (UTC), the
# display as detected (640x480, 256 colours) after its test.
#
# Each step goes on as soon as its screen shows: text screens are waited for
# by their words, graphical ones by a region only that screen has (wait
# screen x y w h digest). The regions were picked from a run's frames, one
# every 500 ms: steady while the screen showed, and nowhere else in the run
# (screen_digest.py prints a region's digest for screenshots). The boot
# menu's 30 s countdown is cut short with Enter.
cd load 0 /mnt/iso/os/Windows NT/Microsoft Windows NT 4.0 Workstation (4.00.1381.1.sp1)/winnt40wks_sp1_en.iso
mark textmode
wait text 120000 To set up Windows NT now, press ENTER
key enter
wait text 60000 Setup has recognized the following mass storage devices
key enter
# The disk is empty: "new or has been erased"
wait text 60000 To continue Setup, press C
key c
# The licence, paged to its end
wait text 60000 PAGE DOWN=Next Page
key pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn pgdn
wait text 30000 F8=I agree
key f8
wait text 60000 The above list matches my computer
key enter
# The whole disk (4095 MB), NTFS (formatted FAT now, converted at the next
# boot), \WINNT
wait text 60000 Unpartitioned space
key enter
wait text 60000 Format the partition using the NTFS file system
key down enter
wait text 600000 Setup installs Windows NT files onto your hard disk
key enter
# The exhaustive examination too, then the copy
wait text 60000 Setup will now examine your hard disk(s) for corruption
key enter
wait text 1800000 This portion of Setup has completed successfully
# Restart, the CD out first, as it boots before the hard disk. The first
# boot converts the partition to NTFS and restarts; the CD goes back in at
# the second boot's menu, when the BIOS has gone past it
mark restart
cd eject 0
key enter
wait text 180000 Converting file system
wait text 120000 Seconds until highlighted choice
